import 'package:flutter/material.dart';
import 'package:google_fonts/google_fonts.dart';

class AppTheme {
  // Light Theme
  static ThemeData get light {
    return ThemeData(
      useMaterial3: true,
      brightness: Brightness.light,
      colorSchemeSeed: Colors.cyan,
      textTheme: GoogleFonts.interTextTheme(ThemeData.light().textTheme),
    );
  }

  // Dark "Galaxy" Theme
  static ThemeData get dark {
    return ThemeData(
      useMaterial3: true,
      brightness: Brightness.dark,
      scaffoldBackgroundColor: const Color(0xFF0B0E14), // Deep space black
      colorScheme: ColorScheme.fromSeed(
        seedColor: Colors.cyanAccent,
        brightness: Brightness.dark,
        surface: const Color(0xFF1A1F26), // Card color
      ),
      // This line replaces those 500 lines of manual font styling!
      textTheme: GoogleFonts.interTextTheme(ThemeData.dark().textTheme),
      
      // Adding a nice card theme for your galaxy look
      cardTheme: CardTheme(
        color: const Color(0xFF1A1F26),
        elevation: 0,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
      ),
    );
  }
}
